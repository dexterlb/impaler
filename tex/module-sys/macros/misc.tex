% --- Listings ---------------------------------------------------------------
\lstdefinelanguage{ild}{
  sensitive=true,
  morecomment=[l]{;},
  morestring=[b]",
}
\lstdefinestyle{ildstyle}{
  language=ild,
  basicstyle=\ttfamily\normalsize,
  keywordstyle=,
  commentstyle=\color{clistingcomment},
  stringstyle=\color{chost},
  frame=none,
  xleftmargin=6pt, xrightmargin=6pt,
  aboveskip=0pt, belowskip=0pt,
  columns=fullflexible,
  keepspaces=true,
  showstringspaces=false,
}
\lstnewenvironment{ildcode}[1][]{\lstset{style=ildstyle,#1}}{}
\lstnewenvironment{schemecode}[1][]{\lstset{style=ildstyle,language=Lisp,#1}}{}

% Listings float on one continuous mdframed rectangle; listings' own per-line
% fill can leave hairline gaps between lines.
\mdfdefinestyle{listingframe}{
  hidealllines=true,
  backgroundcolor=listingfill,
  innerleftmargin=0pt, innerrightmargin=0pt,
  innertopmargin=4pt, innerbottommargin=4pt,
  skipabove=6pt, skipbelow=2pt,
}
\surroundwithmdframed[style=listingframe]{ildcode}
\surroundwithmdframed[style=listingframe]{schemecode}

\newcounter{listing}
\newcommand{\lstcaption}[1]{%
  \par\nobreak\smallskip\noindent
  \refstepcounter{listing}\textbf{Listing \thelisting:} #1\par\medskip
}

% --- Full-width floats ------------------------------------------------------
% \begin{cfig}{<name>}{<caption>} ... \end{cfig}: a two-column-spanning float
% captioned "<name> <n>: <caption>".
\newenvironment{cfig}[2]{%
  \begin{figure*}[t]
  \centering
  \refstepcounter{figure}%
  \def\cfigname{#1}%
  \def\cfigcaption{#2}%
}{%
  \addtocounter{figure}{-1}%
  \captionsetup{name={\cfigname},labelsep=colon}%
  \caption{\cfigcaption}%
  \end{figure*}%
}
