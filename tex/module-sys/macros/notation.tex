% lists
\makeatletter
% \spacedlist[<tail>]{<item>}{<item>}... : variadic; separates the items with a
% space, then expands <tail> once no further brace group follows.
\newcommand{\spacedlist}{\@ifnextchar[{\spacedlist@tail}{\spacedlist@tail[]}}
\def\spacedlist@tail[#1]{\spacedlist@go{#1}}
\def\spacedlist@go#1#2{#2\spacedlist@loop{#1}}
\def\spacedlist@loop#1{\@ifnextchar\bgroup{\spacedlist@item{#1}}{#1}}
\def\spacedlist@item#1#2{\ #2\spacedlist@loop{#1}}
\makeatother

% full-width spans
\newenvironment{cfig}[4]{%
  \begin{figure*}[t]
  \centering
  \Description{#4}%
  \def\cfigname{#1}%
  \def\cfigcaption{#3}%
  \refstepcounter{#2}%
  \edef\cfigprintnum{\csname the#2\endcsname}%
}{%
  \captionsetup{name={\cfigname},labelsep=colon}%
  \def\thefigure{\cfigprintnum}%
  \addtocounter{figure}{-1}%
  \caption{\cfigcaption}%
  \end{figure*}%
}
