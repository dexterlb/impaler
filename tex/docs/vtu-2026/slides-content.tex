\section{Background piece 1: the REPL}

\begin{frame}[fragile]{The REPL}
\note{
The first lisps were built with a REPL in mind: the program lives
in the machine's memory and is incrementally grown by user statements
such as define
}
\begin{listingbox}\begin{schemerepl}
#;1> (+ 3 5)
8

#;2> (define pi 3.14159)

#;3> (* pi 2)
6.28318

#;4> (circumference 42)
Error: unbound variable: circumference

#;5> (define (circumference radius) (* 2 pi radius))

#;6> (circumference 42)
263.89356
\end{schemerepl}\end{listingbox}
\lstcaption{Using the REPL in Scheme}
\end{frame}

\begin{frame}[fragile]{The REPL: Scheme programs}
\note{
Modern programming: we store the program as code on disk, and then
evaluate it before starting the repl. Every time we modify the program,
we start a new interpreter instance and forget the state of the old program.
}
\begin{listingbox}\begin{schemecode}
(define (even? x)
  (if (= x 0)
    #t
    (odd? (- x 1))))

(define (odd? x)
  (if (= x 1)
    #t
    (even? (- x 1))))
\end{schemecode}\end{listingbox}
\lstcaption{A program, e.g. \pathcode{even-odd.scm}}
\begin{listingbox}\begin{schemerepl}
#;1> (load "even-odd.scm")

#;2> (even? 42)
#t
\end{schemerepl}\end{listingbox}
\lstcaption{Using a function defined by the program}
\end{frame}

\begin{frame}[fragile]{(Not) The REPL: Scheme programs}
\note{
Real-world programs do not use the REPL at all, and are evaluated once.
However, LISP-like languages are still built with the
``mutate a global env with statements''
paradigm in mind.
}
\begin{listingbox}\begin{schemecode}
(define (even? x)
  (if (= x 0)
    #t
    (odd? (- x 1))))

(define (odd? x)
  (if (= x 1)
    #t
    (even? (- x 1))))

(define (main)
  (display "Number: ")
  (let ((n (read)))
    (display (if (even? n) "Even" "Odd"))
    (newline)))

(main)
\end{schemecode}\end{listingbox}
\lstcaption{A program that interacts with the user}
\end{frame}

\begin{frame}[fragile]{Definitions are implicit mutations}
\note{
...However, LISP-like languages, although \textbf{functional},
are still built with the
``mutate a global env with statements''
paradigm in mind.
}
\begin{listingbox}\begin{schemecode}
(set! even?
  (lambda (x)
    (if (= x 0)
      #t
      (odd? (- x 1)))))

(set! odd?
  (lambda (x)
    (if (= x 1)
      #t
      (even? (- x 1)))))

(set! main
  (lambda ()
    (display "Number: ")
    (let ((n (read)))
      (display (if (even? n) "Even" "Odd"))
      (newline))))
\end{schemecode}\end{listingbox}
\lstcaption{We unmask ``define'' for what it really is: ``set!''}
\end{frame}

\begin{frame}[fragile]{Static scoping}
\note{
This statically-scoped program resembles purely-functional
languages more closely.
}
\begin{listingbox}\begin{schemecode}
(letrec
  ( (even? (lambda (x)
             (if (= x 0)
               #t
               (odd? (- x 1)))))

    (odd? (lambda (x)
            (if (= x 1)
              #t
              (even? (- x 1)))))

    (main (lambda ()
            (display "Number: ")
            (let ((n (read)))
              (display (if (even? n) "Even" "Odd"))
              (newline)))))
  (main))
\end{schemecode}\end{listingbox}
\lstcaption{The same program, statically-scoped}
\end{frame}

\begin{frame}{Mutating vs statically-scoped programs}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.4\textwidth}
\begin{itemize}
\item Mutable environments
\item Series of statements
\item Mixes functional code and imperative constructs
\end{itemize}
\end{column}
\begin{column}{0.02\textwidth}
\centering
\rule{0.4pt}{0.225\textheight}
\end{column}
\begin{column}{0.4\textwidth}
\begin{itemize}
\item Fully immutable
\item Nested, statically-scoped expressions
\item Declarative
\item Purely functional
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\section{Background piece 2: macros}

\begin{frame}[fragile]{Tracing evaluation}
In the following examples, we will use a \pathcode{trace} helper:
\begin{listingbox}\begin{schemecode}
(define (trace name value)
  (display name)
  (display ": ")
  (display value)
  (newline)
  value)
\end{schemecode}\end{listingbox}
\lstcaption{A \pathcode{trace} function helps us see what is evaluated}
\begin{listingbox}\begin{schemerepl}
#;1> (trace 'value 42)
value: 42
42
\end{schemerepl}\end{listingbox}
\lstcaption{\pathcode{trace} prints its argument and returns it}
\end{frame}

\begin{frame}[fragile]{Scheme example: Boolean AND}
\note{
  This AND function returns the second argument if the first
  is false, and false otherwise. Although ``if'' evals only
  one branch, both arguments to ``and'' are evaled because it
  is a function, duh.
}
\begin{listingbox}\begin{schemecode}
(define (and a b)
  (if a b #f))
\end{schemecode}\end{listingbox}
\lstcaption{Boolean AND as a function}
\begin{listingbox}\begin{schemerepl}
#;1> (and (trace 'left #t) (trace 'right #f))
left: #t
right: #f
#f

#;2> (and (trace 'left #f) (trace 'right #t))
left: #f
right: #t
#f
\end{schemerepl}\end{listingbox}
\lstcaption{The function evaluates both arguments}
\end{frame}
\begin{frame}[fragile]{Scheme macro example: Boolean AND}
\note{
The macro does not eval arguments and just transforms the code
before it is evaled. Tell the audience that \code{(and a b)}
becomes \code{(if a b \#f)}
}
\begin{listingbox}\begin{schemecode}
(define-syntax and
  (syntax-rules ()
    ((and a b)
     (if a b #f))))
\end{schemecode}\end{listingbox}
\lstcaption{Defining ``and'' as a macro}
\begin{listingbox}\begin{schemerepl}
#;1> (and (trace 'left #t) (trace 'right #f))
left: #t
right: #f
#f

#;2> (and (trace 'left #f) (trace 'right #t))
left: #f
#f
\end{schemerepl}\end{listingbox}
\lstcaption{Short-circuiting: the second argument is not evaluated}
\end{frame}

\begin{frame}[fragile]{LISP macro example: boolean AND}
\note{
LISP macros are simply programs that return other programs.
This approach is simpler and more flexible, but has drawbacks
which we will not discuss here (hygiene).
}
\begin{listingbox}\begin{schemecode}
(defmacro and (a b)
  (cons 'if (cons a (cons b (cons nil nil)))))
\end{schemecode}\end{listingbox}
\lstcaption{An unhygienic LISP version of ``and''}
\vspace{1em}
\begin{listingbox}\begin{schemerepl}
CL-USER> (and (trace 'left t) (trace 'right nil))
LEFT: T
RIGHT: NIL
NIL

CL-USER> (and (trace 'left nil) (trace 'right t))
LEFT: NIL
NIL
\end{schemerepl}\end{listingbox}
\lstcaption{Short-circuiting: the second argument is not evaluated}
\end{frame}

\begin{frame}{Macros and static scoping}
\begin{itemize}
\item Macro expansion is a separate, pre-evaluation step
\item \pathcode{defmacro} and \pathcode{define-syntax} have statement-like semantics in a global scope
\begin{itemize}
  \item Locally-scoped macro binding constructs exist but even then macros are in
        a different namespace from the rest of the program and scoping is a syntactic illusion
\end{itemize}
\item Macros are not first-class objects
\end{itemize}
\end{frame}

\section{The language \lang}

\begin{frame}[fragile]{Goals}
\begin{itemize}
    \item First-class macros
    \item Statically-scoped programs, including macro invocations
    \item Minimality of the core language
        \begin{itemize}
            \item Only 3 special forms: \linline|quote|, and \linline|macroexpand| and \linline|free-vars|
            \item Implement everything else in \lang itself
        \end{itemize}
    \item Expressiveness
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Example high-level code in \lang}
\note{
Notice the defs block: it says ``here are the definitions''.
It syntactically resembles the list of defines but semantically
it is statically scoped because of how it is interpreted (we will see later).
}
\begin{listingbox}\begin{ildcode}
(module
  (doc "this module calculates 5!")
  (exports main)
  (imports
    (host (macroexpand <= * +))
    ("core/prelude.ild" (if))
    ("core/module-utils.ild" (fn)))
  (defs
    (!fn main () (fact 5))

    (!fn fact (x)
      (!if (<= x 0)
        1
        (* x (fact (+ x -1)))))))
\end{ildcode}\end{listingbox}
\end{frame}

\begin{frame}[fragile]{Builtin special forms in \lang}
\begin{itemize}
  \item \linline|free-vars| -- introspect the environment (frame) at the callsite
  \item \linline|quote| -- return the given expression without evaluating
  \begin{itemize}
    \item \linline|'<expr>| \(\mapsto\) \linline|(quote <expr>)|
  \end{itemize}
  \item \linline|macroexpand| -- apply given macro to quoted arguments and then evaluate result
  \begin{itemize}
    \item \linline|(!<expr1> ... <exprN>)| \(\mapsto\) \linline|(macroexpand <expr1> ... <exprN>)|
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Requirements for bootstrapping}
\note{
So, how do we get from the special forms [on the prev slide] to the
modular program 2 slides ago? this
}
Functions and values assumed to be included in the initial environment:
\begin{itemize}
\item Numbers: \linline|+|, \linline|*|, \linline|-|, \linline|/|, \linline|=|, \linline|<|, \linline|<=|, \linline|>=|
\item Pairs: \linline|cons|, \linline|car|, \linline|cdr|
\item Predicates: \linline|null?|, \linline|pair?|, \linline|symbol?|, \linline|string?|, \linline|func?|, \linline|fail?|
\item Equality: \linline|sym-eq?|, \linline|str-eq?|
\item Failure: \linline|make-fail|
\item Booleans: \tinline|#t|, \tinline|#f|, \linline|bool-to-k|
\item Special forms: \linline|quote|, \linline|macroexpand|, \linline|free-vars|
\item Abstractions: \linline|mk-lambda|, \linline|poly-fix|
\item Interpreter: \linline|eval|, \linline|apply|, \linline|read-source|
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Macro execution in \lang}
\note{
\begin{itemize}
\item There is no separate macro expansion step.
\item ``runtime'' macro semantics
\end{itemize}
}
This code:
\begin{listingbox}\begin{ildcode}
(!if (= x 5)
  "x is 5"
  "x is not 5")
\end{ildcode}\end{listingbox}
Is semantically equivalent to:
\begin{listingbox}\begin{ildcode}
(eval (free-vars) (if (quote (= x 5))
  (quote "x is 5")
  (quote "x is not 5")))
\end{ildcode}\end{listingbox}
Macros are just functions that operate on code.
\end{frame}

\begin{frame}[fragile]{Defining the building blocks of a module system}
Bootstrapping can be done in a few tiers:
\begin{itemize}
    \item \linline|lambda|
    \item Basic helpers: \linline|list|, \linline|cadr|, \linline|caar|
    \item \linline|if| (via \linline|bool-to-k|)
    \item \linline|map| (via \linline|poly-fix|)
    \item \linline|let|
    \item \linline|let*|
    \item \linline|letrec| (via \linline|poly-fix|)
    \item Operations with key-value lists
    \item Module parsing
    \item Imports
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Bootstrapping specific constructs}
\begin{listingbox}\begin{ildcode}
; definition of "lambda"
(mk-lambda
  (free-vars)
  '(arg-names body)
  '(cons mk-lambda
    (cons (cons free-vars '())
      (cons (cons quote (cons arg-names '()))
        (cons (cons quote (cons body '()))
          '())))))

; definition of "Y" as a special case of poly-fix
(!lambda (f) (car (poly-fix f)))
\end{ildcode}\end{listingbox}
\end{frame}

\begin{frame}[fragile]{Bootstrapping specific constructs}
\note{
We will use ``map'' in the implementation of ``letrec'',
so we need to define it via ``Y''
}
\begin{listingbox}\begin{ildcode}
; definition of "map" without use of letrec
(Y (!lambda (map)
  (!lambda (f l)
    (!if (null? l)
      l
      (!if (pair? l)
        (cons
          (f (car l)) (map f (cdr l)))
        (make-fail
          (list 'not-a-list l)))))))
\end{ildcode}\end{listingbox}
\end{frame}

\begin{frame}[fragile]{Bootstrapping specific constructs}
\note{
\begin{itemize}
\item ``let'' is transformed to \textbf{the result} of a lambda expression
\item here we use ``lambda'' as a regular function - discuss!
\end{itemize}
}
We can now define ``let'':
\begin{listingbox}\begin{ildcode}
; definition of "let"
(!lambda (letlist body)
  (cons
    (lambda
      (map car letlist)
      body)
    (map cadr letlist)))
\end{ildcode}\end{listingbox}

Which, when evaluated, lets us write familiar ``let'' expressions:
\begin{columns}[c,onlytextwidth]
\begin{column}{0.45\textwidth}
\begin{listingbox}\begin{ildcode}
(!let (
    (x 5)
    (y 10))
  (+ x y))
\end{ildcode}\end{listingbox}
\vspace{2pt}
\par\noindent\hfill\rotatebox{90}{$\approx$}\hfill\null\par
\begin{listingbox}\begin{ildcode}
((!lambda (x y)
    (+ x y))
  5 10)
\end{ildcode}\end{listingbox}
\end{column}
\begin{column}{0.06\textwidth}
\centering
$\Rightarrow$
\end{column}
\begin{column}{0.45\textwidth}
\begin{listingbox}\begin{ildcode}
((mk-lambda
    (free-vars)
    '(x y)
    '(+ x y))
  5 10)
\end{ildcode}\end{listingbox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Back to the module example}
\note{
In a similar fashion to how we defined letrec, we define the rest of the
owl (the module loader) to allow us to convert such modules to letrecs
}
\begin{listingbox}\begin{ildcode}
(module
  (doc "this module calculates 5!")
  (exports main)
  (imports ...)
  (defs
    (!fn main () (fact 5))

    (!fn fact (x)
      (!if (<= x 0)
        1
        (* x (fact (+ x -1)))))))
\end{ildcode}\end{listingbox}
\par\noindent\hfill\rotatebox{-90}{$\Rightarrow$}\hfill\null\par
\begin{listingbox}\begin{ildcode}
(eval <env containing imports> (letrec
    ( (main (!lambda () (fact 5)))
      (fact (!lambda (x)
              (!if (<= x 0)
                1
                (* x (fact (+ x -1)))))) )
  (list (cons 'main main))))
\end{ildcode}\end{listingbox}
\end{frame}

\begin{frame}[fragile]{Back to the module example}
\begin{listingbox}\begin{ildcode}
(module
  (doc "this module calculates 5!")
  (exports main)
  (imports
    (host (macroexpand <= * +))
    ("core/prelude.ild" (if))
    ("core/module-utils.ild" (fn)))
  (defs
    (!fn main () (fact 5))

    (!fn fact (x)
      (!if (<= x 0)
        1
        (* x (fact (+ x -1)))))))
\end{ildcode}\end{listingbox}
\begin{itemize}
    \item Modules are converted to \linline|letrec| definitions that allow mutual recursion
    \item The result of evaluating these definitions is a key-value list of exported items
    \item Importing modules works by first evaluating them and then making their exports available in the new environment
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Using modular programs}
\note{
The ILD evaluator contains no magic for supporting modules.
To execute the example program, we simply evaluate the module loader
and apply it to the program filename and the entrypoint function name,
and it does all the work.
}
\begin{listingbox}\begin{ildcode}
((eval (free-vars) (read-source
  "core/bootstrap/module-loader.ild"))

  "demos/fact.ild"
  'main)
\end{ildcode}\end{listingbox}
\lstcaption{Executing the module loader on a user program}
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}
  \item The core of \lang is minimal and purely functional
  \item Unlike traditional lisps, \lang programs are fully statically-scoped
  \item All metaprogramming up to module loading and imports can be written in \lang itself
\end{itemize}
\end{frame}
